### Adım 1 — CSS
```css
.sey-sheet-in { animation: sey-sheet-in var(--dur-4) var(--ease-spring); }
.sey-sheet-out{ animation: sey-sheet-out var(--dur-3) var(--ease-in) forwards; }
@keyframes sey-sheet-in { from{ transform:translateY(100%); opacity:.4 } to{ transform:none; opacity:1 } }
@keyframes sey-sheet-out{ to  { transform:translateY(100%); opacity:0 } }
.sey-backdrop-out{ animation: sey-backdrop-out var(--dur-3) var(--ease-in) forwards; }
@keyframes sey-backdrop-out{ to { opacity:0 } }
@media (prefers-reduced-motion: reduce){
  .sey-sheet-in,.sey-sheet-out,.sey-backdrop-out{ animation:none!important }
}
```

### Adım 2 — Ortak yardımcı (`mediaFx.js`, `SeyFx` üzerine)
```js
    sheetClose: function(cardId, backdropId, done){
      if (typeof done !== 'function') return;
      if (!isPremiumFxEnabled()){ done(); return; }      // premium kapalı VEYA reduced-motion
      var card = document.getElementById(cardId);
      var back = backdropId ? document.getElementById(backdropId) : null;
      if (!card){ done(); return; }
      var fired = false;
      function go(){ if (fired) return; fired = true; done(); }
      card.classList.add('sey-sheet-out');
      if (back) back.classList.add('sey-backdrop-out');
      card.addEventListener('animationend', go, { once: true });
      setTimeout(go, 260);                                // AĞ
    },
```
